cmake_minimum_required(VERSION 3.25)
project(NordSSHNative VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

option(NORDSSH_BUILD_VT_GATE "Build pinned Ghostty SDK C++ ABI engineering test" OFF)
option(NORDSSH_BUILD_TERMINAL "Build native Qt raster terminal engineering gate" OFF)
set(NORDSSH_GHOSTTY_SDK "" CACHE PATH "Hash/commit-verified Ghostty VT SDK prefix")
if(NORDSSH_BUILD_VT_GATE OR NORDSSH_BUILD_TERMINAL)
    find_path(NORDSSH_GHOSTTY_INCLUDE ghostty/vt.h PATHS "${NORDSSH_GHOSTTY_SDK}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(NORDSSH_GHOSTTY_STATIC NAMES ghostty-vt-static PATHS "${NORDSSH_GHOSTTY_SDK}/lib" NO_DEFAULT_PATH REQUIRED)
endif()
if(NORDSSH_BUILD_VT_GATE AND BUILD_TESTING)
    add_executable(ghostty_abi_tests tests/GhosttyAbiTests.cpp)
    target_include_directories(ghostty_abi_tests PRIVATE "${NORDSSH_GHOSTTY_INCLUDE}")
    target_compile_definitions(ghostty_abi_tests PRIVATE GHOSTTY_STATIC)
    target_link_libraries(ghostty_abi_tests PRIVATE "${NORDSSH_GHOSTTY_STATIC}")
    if(WIN32)
        target_link_libraries(ghostty_abi_tests PRIVATE ntdll kernel32)
    endif()
    add_test(NAME ghostty_abi COMMAND ghostty_abi_tests)
    set_tests_properties(ghostty_abi PROPERTIES TIMEOUT 30)
endif()

option(NORDSSH_BUILD_CRYPTO "Build authenticated vault envelope gate" OFF)
if(NORDSSH_BUILD_CRYPTO)
    find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS Crypto)
    include(cmake/Argon2.cmake)
    add_library(nordssh_vault_crypto STATIC src/vault/VaultCrypto.cpp)
    target_include_directories(nordssh_vault_crypto PUBLIC src)
    target_link_libraries(nordssh_vault_crypto PUBLIC nordssh_vault_format OpenSSL::Crypto Argon2::argon2)
    if(MSVC)
        target_compile_options(nordssh_vault_crypto PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(nordssh_vault_crypto PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BUILD_TESTING)
        add_executable(vault_crypto_tests tests/VaultCryptoTests.cpp)
        target_link_libraries(vault_crypto_tests PRIVATE nordssh_vault_crypto)
        add_test(NAME vault_crypto COMMAND vault_crypto_tests)
        set_tests_properties(vault_crypto PROPERTIES TIMEOUT 60)
    endif()
endif()

option(NORDSSH_BUILD_SSH "Build isolated SSH transport gate" OFF)
if(NORDSSH_BUILD_SSH)
    if(NOT NORDSSH_BUILD_CRYPTO)
        message(FATAL_ERROR "SSH gate requires NORDSSH_BUILD_CRYPTO")
    endif()
    find_package(libssh 0.12.2 EXACT CONFIG REQUIRED)
    add_library(nordssh_ssh STATIC src/ssh/Session.cpp src/ssh/KeyMaterial.cpp)
    target_include_directories(nordssh_ssh PUBLIC src)
    target_link_libraries(nordssh_ssh PUBLIC nordssh_vault_crypto ssh)
    if(MSVC)
        target_compile_options(nordssh_ssh PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(nordssh_ssh PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BUILD_TESTING)
        add_executable(ssh_transport_tests tests/SshTransportTests.cpp)
        target_link_libraries(ssh_transport_tests PRIVATE nordssh_ssh)
        # Executed only through the loopback fixture, not against real hosts.
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME ssh_transport COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/ssh_fixture.py $<TARGET_FILE:ssh_transport_tests>)
        set_tests_properties(ssh_transport PROPERTIES TIMEOUT 45)
        add_executable(ssh_key_tests tests/SshKeyTests.cpp)
        target_link_libraries(ssh_key_tests PRIVATE nordssh_ssh)
        add_test(NAME ssh_keys COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/ssh_key_fixture.py $<TARGET_FILE:ssh_key_tests>)
        set_tests_properties(ssh_keys PROPERTIES TIMEOUT 90)
    endif()
endif()

add_library(nordssh_vault_format STATIC src/vault/VaultFormat.cpp src/vault/PayloadAdmission.cpp)
target_include_directories(nordssh_vault_format PUBLIC src)
if(MSVC)
    target_compile_options(nordssh_vault_format PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(nordssh_vault_format PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

option(NORDSSH_BUILD_QT_INPUT "Build native Qt input routing gate" OFF)
if(NORDSSH_BUILD_QT_INPUT OR NORDSSH_BUILD_TERMINAL)
    find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Widgets Test)
    add_library(nordssh_input STATIC src/input/PasteRouter.cpp)
    target_include_directories(nordssh_input PUBLIC src)
    target_link_libraries(nordssh_input PUBLIC Qt6::Widgets)
    if(BUILD_TESTING)
        add_executable(paste_router_tests tests/PasteRouterTests.cpp)
        target_link_libraries(paste_router_tests PRIVATE nordssh_input Qt6::Test)
        add_test(NAME paste_router COMMAND paste_router_tests)
        set_tests_properties(paste_router PROPERTIES TIMEOUT 10 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
if(NORDSSH_BUILD_TERMINAL)
    add_library(nordssh_terminal STATIC src/terminal/TerminalModel.cpp src/terminal/TerminalWidget.cpp)
    target_include_directories(nordssh_terminal PUBLIC src PRIVATE "${NORDSSH_GHOSTTY_INCLUDE}")
    target_compile_definitions(nordssh_terminal PRIVATE GHOSTTY_STATIC)
    target_link_libraries(nordssh_terminal PUBLIC nordssh_input Qt6::Widgets PRIVATE "${NORDSSH_GHOSTTY_STATIC}")
    if(WIN32)
        target_link_libraries(nordssh_terminal PRIVATE ntdll kernel32)
    endif()
    if(MSVC)
        target_compile_options(nordssh_terminal PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(nordssh_terminal PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BUILD_TESTING)
        add_executable(terminal_widget_tests tests/TerminalWidgetTests.cpp)
        add_executable(terminal_history_tests tests/TerminalHistoryTests.cpp)
        add_executable(terminal_mouse_tests tests/TerminalMouseTests.cpp)
        target_link_libraries(terminal_widget_tests PRIVATE nordssh_terminal Qt6::Test)
        target_link_libraries(terminal_history_tests PRIVATE nordssh_terminal)
        target_link_libraries(terminal_mouse_tests PRIVATE nordssh_terminal)
        if(MSVC)
            target_compile_options(terminal_widget_tests PRIVATE /W4 /WX /utf-8)
            target_compile_options(terminal_history_tests PRIVATE /W4 /WX /utf-8)
            target_compile_options(terminal_mouse_tests PRIVATE /W4 /WX /utf-8)
        else()
            target_compile_options(terminal_widget_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
            target_compile_options(terminal_history_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
            target_compile_options(terminal_mouse_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
        add_test(NAME terminal_widget COMMAND terminal_widget_tests)
        add_test(NAME terminal_history COMMAND terminal_history_tests)
        add_test(NAME terminal_mouse COMMAND terminal_mouse_tests)
        set_tests_properties(terminal_mouse PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        set_tests_properties(terminal_history PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        set_tests_properties(terminal_widget PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        if(WIN32)
            set_property(TEST terminal_widget APPEND PROPERTY ENVIRONMENT "NORDSSH_TEST_FONT=$ENV{SystemRoot}/Fonts/consola.ttf")
        endif()
        if(NORDSSH_BUILD_SSH)
            add_executable(terminal_ssh_tests tests/TerminalSshTests.cpp)
            target_link_libraries(terminal_ssh_tests PRIVATE nordssh_terminal nordssh_ssh)
            if(MSVC)
                target_compile_options(terminal_ssh_tests PRIVATE /W4 /WX /utf-8)
            else()
                target_compile_options(terminal_ssh_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
            endif()
            add_test(NAME terminal_ssh COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/ssh_fixture.py $<TARGET_FILE:terminal_ssh_tests> --terminal)
            set_tests_properties(terminal_ssh PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        endif()
    endif()
endif()
if(BUILD_TESTING)
    add_executable(vault_format_tests tests/VaultFormatTests.cpp)
    add_executable(payload_admission_tests tests/PayloadAdmissionTests.cpp)
    target_link_libraries(payload_admission_tests PRIVATE nordssh_vault_format)
    add_test(NAME payload_admission COMMAND payload_admission_tests)
    set_tests_properties(payload_admission PROPERTIES TIMEOUT 20)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME payload_oracle COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/payload_oracle.py $<TARGET_FILE:payload_admission_tests>)
        set_tests_properties(payload_oracle PROPERTIES TIMEOUT 30)
    endif()
    target_link_libraries(vault_format_tests PRIVATE nordssh_vault_format)
    add_test(NAME vault_format COMMAND vault_format_tests)
    set_tests_properties(vault_format PROPERTIES TIMEOUT 10)
endif()
